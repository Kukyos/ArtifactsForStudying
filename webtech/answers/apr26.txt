```run html
  Annual Interest(%): <input type="number" ng-model="rate"><br><br>
  Tenure(Months): <input type="number" ng-model="n"><br><br>
  <button ng-click="calculate()">Calculate EMI</button>

  <div ng-show="emi">
    <p><b>EMI: Rs.{{ emi }} per month<br>
    Total Repayment: Rs.{{ total }}<br>
    Total Interest: Rs.{{ interest }}</b></p>
  </div>

  <script>
    angular.module("emiApp", []).controller("emiCtrl", function ($scope) {
      $scope.p = 100000; $scope.rate = 10.99; $scope.n = 12;
      $scope.calculate = function () {
        var r = $scope.rate / 12 / 100;               // monthly rate
        var x = Math.pow(1 + r, $scope.n);
        $scope.emi = Math.round($scope.p * r * x / (x - 1));   // E = p*r*(1+r)^n / ((1+r)^n - 1)
        $scope.total = $scope.emi * $scope.n;
        $scope.interest = $scope.total - $scope.p;
      };
    });
  </script>
</body>
</html>
```
<p>The inputs are bound to scope variables with <code>ng-model</code>. <code>ng-click</code> runs <code>calculate()</code>, which converts the yearly rate to a monthly rate r and applies the formula. The EMI is rounded to whole rupees (as in the paper's sample); total repayment = EMI × n and total interest = total − principal. With the paper's input (Rs.100000, 10.99%, 12 months) it gives exactly the sample: EMI Rs.8838, total Rs.106056, interest Rs.6056.</p>

<p><b>Routing</b> makes a single-page application (SPA): the page loads once, and only the part inside <code>ng-view</code> is swapped when a link is clicked. It needs the <b>ngRoute</b> module (<code>angular-route.js</code>), which is added as a dependency. <code>$routeProvider.when(path, {template or templateUrl})</code> maps each URL path to a view, and <code>otherwise()</code> is the default. Links use <code>#!/path</code>.</p>
```run html
<!DOCTYPE html>
<html>

  <script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.8.2/angular.min.js"></script>
  <script src="https://ajax.googleapis.com/ajax/libs/angularjs/1.8.2/angular-route.min.js"></script>
</head>
<body ng-app="myApp">
  <h2>Home Page</h2>
  <a href="#!login">Login</a> |
  <a href="#!signup">New User Sign Up</a>
  <hr>
  <div ng-view></div>

  <script>
    var app = angular.module("myApp", ["ngRoute"]);
    app.config(function ($routeProvider) {
      $routeProvider
        .when("/login", {
          template: "<h3>Login</h3>Username: <input><br>Password: <input type='password'><br><button>Login</button>"
        })
        .when("/signup", {
          template: "<h3>New User Sign Up</h3>Name: <input><br>Email: <input type='email'><br><button>Sign Up</button>"
        })
        .otherwise({
          template: "<p>Welcome! Choose Login or Sign Up above.</p>"
        });
    });
  </script>
</body>
</html>
```

<p><b>app.js</b></p>
```js
const express = require("express");
const mongoose = require("mongoose");
const path = require("path");

const app = express();
app.use(express.urlencoded({ extended: true }));   // read form data

mongoose.connect("mongodb://127.0.0.1:27017/company")
  .then(() => console.log("Connected to MongoDB"))
  .catch((err) => console.log(err.message));

const userSchema = new mongoose.Schema({
  name: String, email: String, phone: String, state: String
});
const User = mongoose.model("User", userSchema, "users");   // collection "users"

app.get("/", (req, res) => {
  res.sendFile(path.join(__dirname, "index.html"));
});

app.post("/server", async (req, res) => {
  try {
    await new User(req.body).save();
    res.send("<b>Data Saved in MongoDB Database Successfully</b>");
  } catch (err) {
    res.status(500).send("Error saving data: " + err.message);
  }
});

app.listen(2000, () => console.log("Server running at http://localhost:2000/"));
```
