<p>Uses only Node's built-in <b>http</b>, <b>fs</b> and <b>querystring</b> modules: GET <code>/</code> sends payment.html; the form POSTs to <code>/payment</code>, where the body is collected chunk by chunk, parsed, and shown back.</p>
<p><b>payment.html</b></p>
```html
<!DOCTYPE html>
<html>
<body>
  <h2>Payment Gateway</h2>
  <form action="/payment" method="post">
    <table>
      <tr><td>Card Number</td><td><input type="text" name="card"></td></tr>
      <tr><td>Amount</td><td><input type="text" name="amount"></td></tr>
      <tr><td>Account Number</td><td><input type="text" name="accno"></td></tr>
      <tr><td></td><td><br><input type="submit" value="Transfer"></td></tr>
    </table>
  </form>
</body>
</html>
```
<p><b>server.js</b></p>
```js
const http = require("http");
const fs = require("fs");
const qs = require("querystring");

http.createServer((req, res) => {
  if (req.method === "GET" && req.url === "/") {
    fs.readFile("payment.html", (err, data) => {
      res.writeHead(200, { "Content-Type": "text/html" });
      res.end(data);
    });
  } else if (req.method === "POST" && req.url === "/payment") {
    let body = "";
    req.on("data", (chunk) => { body += chunk; });     // collect the posted data
    req.on("end", () => {
      const f = qs.parse(body);                        // "card=..&amount=.." -> object
      res.writeHead(200, { "Content-Type": "text/html" });
      res.end("<div style='border:1px solid black;padding:6px;width:320px'>" +
              "<b>The Transaction Details are:</b><br>" +
              "Card Number: " + f.card + "<br>" +
              "Amount: " + f.amount + "<br>" +
              "Account Number: " + f.accno + "</div>");
    });
  } else {
    res.writeHead(404);
    res.end("Not found");
  }
}).listen(2000, () => console.log("Server running at http://localhost:2000/"));
```
