import pandas as pd

# 1. reading and exploring
df = pd.read_csv('students.csv')
print(df.head())
print(df.tail())
print(df.shape)
print(list(df.columns))
print(df.dtypes)
print(df.describe())

# 2. missing values
print(df.isnull().sum())
df['marks'] = df['marks'].fillna(df['marks'].mean())
df['city'] = df['city'].fillna('unknown')
print(df.isnull().sum())

# 3. filtering
print(df[df['marks'] > 80])
print(df[df['dept'] == 'CSE'])
print(df[(df['dept'] == 'AI') & (df['marks'] > 75)])

# 4. manipulation
df['total marks'] = df['marks'] + df['project']          # marks out of 100 + project out of 20
df['percentage'] = df['total marks'] / 120 * 100
df = df.drop(columns='city')
df = df.rename(columns={'marks': 'score'})
print(df)

# 5. sorting, ranking, grouping
print(df.sort_values(by='score', ascending=False))
df['rank'] = df['score'].rank(ascending=False)
print(df[['name', 'score', 'rank']])
print(df.groupby('dept')['score'].mean())
